cmake_minimum_required(VERSION 3.22.1)
project(openevv LANGUAGES C)

# Set C standard
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Default to release optimization (-O2)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Compiler flags: -O2 ensures bit-exact synthesis across architectures
add_compile_options(
    -O2
    -fomit-frame-pointer
    -DEVV_ARENA=1
    -ffunction-sections
    -fdata-sections
    -fPIC
    -w
    -Wno-implicit-function-declaration
    -Werror=int-conversion
    -Werror=incompatible-pointer-types
)

# Android 15+ 16KB ELF page size alignment
if(ANDROID)
    add_link_options(
        -Wl,-z,max-page-size=16384
        -Wl,-z,common-page-size=16384
        -Wl,--gc-sections
    )
endif()

# Include directories
set(OPENEVV_INCLUDE_DIRS
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_BINARY_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}
)

# Supported bundled languages
set(OPENEVV_LANGS enus dede engb eses esus frca frfr itit plpl)

# Collect core engine C sources
file(GLOB_RECURSE OPENEVV_SRC_FILES "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c")
list(FILTER OPENEVV_SRC_FILES EXCLUDE REGEX ".*/port_win32\\.c$")

# Collect language sources
set(OPENEVV_LANG_FILES "")
foreach(LANG ${OPENEVV_LANGS})
    set(LANG_DIR "${CMAKE_CURRENT_SOURCE_DIR}/lang/${LANG}")
    list(APPEND OPENEVV_INCLUDE_DIRS "${LANG_DIR}")
    file(GLOB LANG_SOURCES "${LANG_DIR}/*.c")
    list(FILTER LANG_SOURCES EXCLUDE REGEX ".*/delta_rules_c.*\\.c$")
    list(APPEND OPENEVV_LANG_FILES ${LANG_SOURCES})
endforeach()

# Generate delta_langs.c listing bound languages
set(DELTA_LANGS_C "${CMAKE_CURRENT_BINARY_DIR}/delta_langs.c")
file(WRITE "${DELTA_LANGS_C}" "/* Generated by CMake */\n#include \"delta_lang.h\"\n\n")
foreach(LANG ${OPENEVV_LANGS})
    file(APPEND "${DELTA_LANGS_C}" "extern delta_language delta_lang_${LANG};\n")
    file(APPEND "${DELTA_LANGS_C}" "void delta_lang_bind_${LANG}(void);\n")
endforeach()
file(APPEND "${DELTA_LANGS_C}" "\nconst delta_language *const delta_languages[] = {\n")
foreach(LANG ${OPENEVV_LANGS})
    file(APPEND "${DELTA_LANGS_C}" "    &delta_lang_${LANG},\n")
endforeach()
file(APPEND "${DELTA_LANGS_C}" "    0,\n};\n\n")
file(APPEND "${DELTA_LANGS_C}" "void delta_lang_bind_all(void)\n{\n")
foreach(LANG ${OPENEVV_LANGS})
    file(APPEND "${DELTA_LANGS_C}" "    delta_lang_bind_${LANG}();\n")
endforeach()
file(APPEND "${DELTA_LANGS_C}" "}\n")

list(APPEND OPENEVV_LANG_FILES "${DELTA_LANGS_C}")

# 1. Target: Shared Library (libopenevv.so)
add_library(openevv SHARED
    ${OPENEVV_SRC_FILES}
    ${OPENEVV_LANG_FILES}
    ${CMAKE_CURRENT_SOURCE_DIR}/lib/eci_api.c
)
target_include_directories(openevv PUBLIC ${OPENEVV_INCLUDE_DIRS})
target_link_libraries(openevv m pthread)
set_target_properties(openevv PROPERTIES OUTPUT_NAME "openevv")

# 2. Target: Standalone CLI executable (evv)
if(NOT DEFINED OPENEVV_BUILD_CLI OR OPENEVV_BUILD_CLI)
    add_executable(evv
        ${OPENEVV_SRC_FILES}
        ${OPENEVV_LANG_FILES}
        ${CMAKE_CURRENT_SOURCE_DIR}/cli/evv.c
    )
    target_include_directories(evv PRIVATE ${OPENEVV_INCLUDE_DIRS})
    target_link_libraries(evv m pthread)
    if(ANDROID)
        set_target_properties(evv PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
    endif()
endif()
